the two ways to write a timeout

// short, and it cannot leak a timer
const signal = AbortSignal.any([
  caller.signal,
  AbortSignal.timeout(2000),
]);
await fetch(url, { signal });

// the long way, which is what an assistant writes
const c = new AbortController();
const t = setTimeout(() => c.abort(), 2000);
try { await fetch(url, { signal: c.signal }); }
finally { clearTimeout(t); }   // <- the line people leave out
